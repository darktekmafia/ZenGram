-- Users and Sessions Table
CREATE TABLE sessions (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id VARCHAR(15) UNIQUE NOT NULL,
    access_token TEXT NOT NULL,
    refresh_token TEXT,
    expires_at TIMESTAMP WITH TIME ZONE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Downloaded Media Metadata (optional but useful for searching/filtering later)
CREATE TABLE downloaded_media (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id VARCHAR(15),  -- Who you follow
    post_id TEXT,         # Instagram's internal ID
    media_url TEXT NOT NULL,
    thumbnail_url TEXT,
    caption TEXT,
    timestamp TIMESTAMP WITH TIME ZONE,
    downloaded_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    file_path VARCHAR(500),  -- Where saved on disk
);

-- User Preferences (filtering rules)
CREATE TABLE user_preferences (
    session_id UUID REFERENCES sessions(id),
    max_posts_per_user INTEGER DEFAULT 10,
    min_likes INT DEFAULT 0,
    include_stories BOOLEAN DEFAULT FALSE,
    PRIMARY KEY (session_id)
);
